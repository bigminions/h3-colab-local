POST /api/jobs 提交 · GET /api/jobs/{id} 查询
GET /api/account 查询剩余单元
POST /api/jobs/{id}/retry 取回 · POST /api/release 释放
import requests, time
base = "http://127.0.0.1:7865"
headers = {"X-H3-Key": "粘贴本机 Key"}
r = requests.post(base + "/api/jobs", headers=headers,
    json={"prompt": "A cat walking in a garden, birdsong.",
          "aspect_ratio": "9:16",
          "people": False, "auto_release": True})
r.raise_for_status()
job_id = r.json()["id"]
while True:
    r = requests.get(f"{base}/api/jobs/{job_id}", headers=headers)
    r.raise_for_status()
    job = r.json()
    if job["status"] in ["done", "warning", "error", "interrupted"]:
        print(job)
        break
    time.sleep(5)
